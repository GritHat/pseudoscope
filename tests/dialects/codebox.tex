\section{Sorting}
\begin{codebox}
\Procname{$\proc{Insertion-Sort}(A)$}
\li \For $j \gets 2$ \To $\attrib{A}{length}$
\li     \Do
            $\id{key} \gets A[j]$
\li         \Comment Insert $A[j]$ into the sorted sequence $A[1 \twodots j-1]$.
\li         $i \gets j-1$
\li         \While $i > 0$ and $A[i] > \id{key}$
\li             \Do
                    $A[i+1] \gets A[i]$
\li                 $i \gets i-1$
                \End
\li         $A[i+1] \gets \id{key}$
        \End
\end{codebox}
\begin{codebox}
\Procname{$\proc{Maximum}(A)$}
\li $\id{max} \gets A[1]$
\li \For $i \gets 2$ \To $\attrib{A}{length}$
\li     \Do \If $A[i] > \id{max}$
\li             \Then $\id{max} \gets A[i]$
\li             \Else $\id{max} \gets \id{max}$
                \End
        \End
\li \Return $\id{max}$
\end{codebox}
\begin{codebox}
\Procname{$\proc{Reach}(G, s)$}
\li $S \gets \{s\}$
\li $Q \gets \emptyset$
\li $\proc{Enqueue}(Q, s)$
\li \While $Q \neq \emptyset$
\li     \Do
            $u \gets \proc{Dequeue}(Q)$
\li         \For each vertex $v \in \attrib{G}{Adj}[u]$
\li             \Do \If $v \notin S$
\li                     \Then $S \gets S \cup \{v\}$
\li                         $\proc{Enqueue}(Q, v)$
                        \End
                \End
        \End
\li \Return $|S|$
\end{codebox}
